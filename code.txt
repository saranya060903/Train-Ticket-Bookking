// ?? ONLY NUMBERS FUNCTION
function onlyNumbers(input){
    input.value = input.value.replace(/[^0-9]/g,'');
}

// USERNAME
function validateUsername(){
    if(!/^[A-Za-z]{6,}$/.test(username.value)){
        userError.innerText="Username must be 6 letters";
        username.classList.add("invalid");
        return false;
    }
    userError.innerText="";
    username.classList.remove("invalid");
    username.classList.add("valid");
    return true;
}

// EMAIL
function validateEmail(){
    if(!email.value.includes("@") || !email.value.includes(".")){
        emailError.innerText="Invalid email";
        email.classList.add("invalid");
        return false;
    }
    emailError.innerText="";
    email.classList.remove("invalid");
    email.classList.add("valid");
    return true;
}

// PASSWORD
function validatePassword(){
    if(!/^(?=.*[A-Z])(?=.*\d)(?=.*[\W]).{8,}$/.test(password.value)){
        passError.innerText="8+ chars, 1 uppercase, number, special";
        password.classList.add("invalid");
        return false;
    }
    passError.innerText="";
    password.classList.remove("invalid");
    password.classList.add("valid");
    return true;
}

// CONFIRM PASSWORD
function validateConfirm(){
    if(password.value !== confirmPassword.value){
        confirmError.innerText="Passwords not match";
        confirmPassword.classList.add("invalid");
        return false;
    }
    confirmError.innerText="";
    confirmPassword.classList.remove("invalid");
    confirmPassword.classList.add("valid");
    return true;
}

// MOBILE
function validateMobile(){
    if(!/^\d{10}$/.test(mobile.value)){
        mobileError.innerText="Must be 10 digits";
        mobile.classList.add("invalid");
        return false;
    }
    mobileError.innerText="";
    mobile.classList.remove("invalid");
    mobile.classList.add("valid");
    return true;
}

// AADHAAR
function validateAadhaar(){
    if(!/^\d{12}$/.test(aadhaar.value)){
        aadhaarError.innerText="Must be 12 digits";
        aadhaar.classList.add("invalid");
        return false;
    }
    aadhaarError.innerText="";
    aadhaar.classList.remove("invalid");
    aadhaar.classList.add("valid");
    return true;
}

// REAL-TIME FOR OTHER FIELDS
username.oninput = validateUsername;
email.oninput = validateEmail;
password.oninput = validatePassword;
confirmPassword.oninput = validateConfirm;

// REGISTER
function register(){

    let isValid = true;

    if(!validateUsername()) isValid=false;
    if(!validateEmail()) isValid=false;
    if(!validatePassword()) isValid=false;
    if(!validateConfirm()) isValid=false;
    if(!validateMobile()) isValid=false;
    if(!validateAadhaar()) isValid=false;

    if(!isValid){
        alert("? Fix all errors first");
        return;
    }

    let userData={
        username:username.value,
        password:password.value,
        email:email.value,
        mobile:mobile.value,
        aadhaar:aadhaar.value
    };

    localStorage.setItem(username.value,JSON.stringify(userData));

    alert("? Registration Successful");
}

function login() {
    let role = document.getElementById("role").value;
    let username = document.getElementById("username").value.trim();
    let password = document.getElementById("password").value;
    let msg = document.getElementById("msg");

    msg.innerText = "";

    if (role === "") {
        msg.innerText = "Please select role";
        return;
    }

    if (username === "" || password === "") {
        msg.innerText = "Username & Password required";
        return;
    }

    // ? ADMIN LOGIN
    // ? ADMIN LOGIN
if (role === "admin") {
    if (password === "admin") {

        // ? Allow ANY username for admin
        localStorage.setItem("currentUser", username);

        window.location.href = "admin.html";

    } else {
        msg.innerText = "? Invalid Admin password";
    }
    return;
}

    // ? CUSTOMER LOGIN
    if (role === "user") {

        let userData = localStorage.getItem(username);

        if (!userData) {
            msg.innerText = "? User not found. Please register!";
            return;
        }

        userData = JSON.parse(userData);

        if (userData.password === password) {
            localStorage.setItem("currentUser", username);

            // ? IMPORTANT CHANGE HERE
            window.location.href = "confirmation.html";

        } else {
            msg.innerText = "? Incorrect password";
        }
    }
}
